from __future__ import annotations

ARGS = ["save-state", "-s", "sess-dup", "--no-lint", "--no-promote", "--no-sync", "-m", "probe"]


def run(memdir):
    return CliRunner().invoke(cli_mod.main, ARGS[:3] + ["--memory-dir", str(memdir)] + ARGS[3:])


def test_p1_flat_pair_fresh_folder_created(memdir):
    kept, other = _pair(memdir, "fresh-folder")
    r = run(memdir)
    print("P1 exit", r.exit_code, r.output)
    top = memdir / f"{DUP}.md"
    print("P1 walk kept before save:", kept, "created kept:", "2026-11-05")
    print("P1 top created after flat save:", fm(top, "created"))
    print("P1 folder still there:", (memdir / "savestates" / f"{DUP}.md").exists())
    print("P1 iter_savestates:", memory_layout.iter_savestates(memdir))

def test_p3_remove_failed(memdir):
    organise(memdir)
    kept, other = _pair(memdir, "fresh-top")
    os.chmod(memdir, 0o555)
    try:
        r = run(memdir)
    finally:
        os.chmod(memdir, 0o755)
    print("P3 exit", r.exit_code, "out:", r.output, repr(r.exception))
    print("P3 top exists:", (memdir / f"{DUP}.md").exists())


@pytest.mark.parametrize("case", ["fresh-folder", "fresh-top"])
def test_p4_partial_write(memdir, monkeypatch, case):
    organise(memdir)
    kept, other = _pair(memdir, case)
    real = Path.write_text

    def half(self, data, *a, **k):
        if self.parent.name == "savestates" and self.name == f"{DUP}.md":
            real(self, data[: len(data) // 3], *a, **k)
            raise OSError(28, "No space left on device")
        return real(self, data, *a, **k)

    monkeypatch.setattr(Path, "write_text", half)
    r = run(memdir)
    monkeypatch.undo()
    print("P4", case, "exit", r.exit_code, "out:", r.output)
    print("P4 iter_savestates:", memory_layout.iter_savestates(memdir))
    lat = handoff._latest_savestate(memdir)
    print("P4 latest path:", lat and lat["path"], "body has FRESH:",
          lat and "FRESH" in lat["body"], "len body:", lat and len(lat["body"]))


def test_p5_verb_raises_verberror(memdir):
    organise(memdir)
    kept, other = _pair(memdir, "tie")
    os.chmod(kept, 0o444)
    os.chmod(kept.parent, 0o555)
    try:
        with pytest.raises(verbs.VerbError) as ei:
            verbs.VERBS["rmx_save_state"].run(
                store_root(), {"session": "sess-dup", "memory_dir": str(memdir),
                               "lint": False, "promote": False, "sync": False})
    finally:
        os.chmod(kept.parent, 0o755)
        os.chmod(kept, 0o644)
    print("P5 VerbError:", ei.value)


def test_p6_remove_failed_uchg(memdir):
    import subprocess
    organise(memdir)
    kept, other = _pair(memdir, "fresh-top")
    subprocess.run(["chflags", "uchg", str(kept)], check=True)
    try:
        r = run(memdir)
    finally:
        subprocess.run(["chflags", "nouchg", str(kept)], check=True)
    print("P6 exit", r.exit_code, "out:", " ".join(r.output.split()), repr(r.exception))
    print("P6 walk:", memory_layout.iter_savestates(memdir))
    lat = handoff._latest_savestate(memdir)
    print("P6 latest has probe msg:", lat and "probe" in lat["body"], lat and lat["path"])
